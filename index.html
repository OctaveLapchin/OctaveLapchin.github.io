# Ce script Powershell permet de déplacer les orinateur d'une OU à l'autre
Get-Content "./postes.csv" | ForEach-Object {
    $nom, $str = ($_ -split ";")[0], ($_ -split ";")[1] -split "-" -replace "\+", "0"
    Get-ADComputer -Identity $nom | Move-ADObject -TargetPath (Get-ADOrganizationalUnit -Filter "Name -eq '$($str[3])'").DistinguishedName
}

function copy(arg) {
    let copyText = document.getElementById(arg);
    if (copyText) {
        const tempInput = document.createElement("textarea");
        tempInput.value = copyText.textContent || copyText.value;
        document.body.appendChild(tempInput);
        tempInput.select();
        document.execCommand("copy");
        document.body.removeChild(tempInput);
        console.log("Texte copié dans le presse-papiers !");
    } else {
        console.error("Élément introuvable !");
    }
}
                                                                                

echo -n "Entrez le sous nom de domain associé au service (ex: glpi.dn.tld)"
read name
echo -n "Entrez le mot de passe de la CA : "
read -s password

cd /etc/pki/private/
openssl genrsa -out $name.key 2048
cd /etc/pki/newcerts/
sudo openssl req -new -key ../private/$name.key -out $name.csr -subj "/C=FR/ST=PACA/L=Avignon/O=ITWay/OU=IT/CN=$name.itway.corp" -addext "subjectAltName=DNS:$name.itway.corp"
echo "subjectAltName=DNS:$name.itway.corp" > ext-$name.cnf
sudo openssl x509 -req -in $name.csr -CA ../certs/ca.crt -CAkey ../private/ca.key -CAcreateserial -out $name.crt -days 365 -extfile ext-$name.cnf -passin pass:$password
# On vérifie que le certificat est bien signé par l autorité de cert
openssl verify -CAfile /etc/pki/certs/ca.crt $name.crt
cp /etc/pki/private/$name.key /home/service/
cp /etc/pki/newcerts/$name.crt /home/service/
chown -R service:service /home/service/*
